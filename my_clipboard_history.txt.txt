def extract_pinterest_links(file_path):
  clean_links = []
  url_pattern = re.compile(r"https://www\.pinterest\.com/pin/\d+/")

  try:
    with open(file_path, mode="r", encoding="utf-8") as f:
      for line in f:
        match = url_pattern.search(line)
        if match:
          clean_url = match.group(0)
          if clean_url not in clean_links:
            clean_links.append(clean_url)
  except FileNotFoundError:
    print(f"❌ Error: '{file_path}' file nahi mili.")

  return clean_links


def get_pinterest_image_url(pin_url):
  try:
    oembed_url = f"https://www.pinterest.com/oembed.json?url={pin_url}"
    res = requests.get(oembed_url, timeout=10)
    if res.status_code == 200:
      data = res.json()
      thumb_url = data.get("thumbnail_url")
      if thumb_url:
        hq_url = thumb_url.replace("/236x/", "/736x/")
        return hq_url
  except Exception:
    pass
  return None


def schedule_facebook_photo(
    page_id, access_token, message, image_url, publish_timestamp
):
  url = f"https://graph.facebook.com/v18.0/{page_id}/photos"

  try:
    img_res = requests.get(image_url, timeout=30)
    if img_res.status_code != 200:
      return {
          "error": {
              "message": f"Image download failed (Status: {img_res.status_code})"
          }
      }
  except Exception as e:
    return {
        "error": {"message": f"Invalid URL or Connection Error: {str(e)}"}
    }

  files = {"source": ("image.jpg", io.BytesIO(img_res.content), "image/jpeg")}

  data = {
      "message": message,
      "published": "false",
      "scheduled_publish_time": publish_timestamp,
      "access_token": access_token,
  }

  response = requests.post(url, data=data, files=files)
  return response.json()

        publish_timestamp=scheduled_time,
    )

    if "id" in result:
      print(f"    ✅ Success! Photo ID: {result['id']}")
      success_count += 1
    else:
      err_msg = result.get("error", {}).get("message", "Unknown Error")
      print(f"    ❌ Error: {err_msg}")
      error_count += 1

    scheduled_time += INTERVAL_HOURS * 3600
    time.sleep(2)

  print(
      f"\n🎉 Batch Complete! Successfully Scheduled Photos: {success_count},"
      f" Failed: {error_count}"
  )


if __name__ == "__main__":
  main()
